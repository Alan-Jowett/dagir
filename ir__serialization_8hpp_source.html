28inline std::string to_json(const dagir::ir_graph& g) {
29 json out;
30 out["schema_version"] = 1;
31
32 // nodes
33 out["nodes"] = json::array();
34 for (const auto& n : g.nodes) {
35 json nn;
36 // Prefer name attribute as id
37 auto it_name = n.attributes.find(dagir::ir_attrs::k_name);
38 if (it_name != n.attributes.end()) {
39 nn["id"] = std::string(it_name->second);
40 } else {
41 nn["id"] = std::to_string(n.id);
42 }
43 if (n.attributes.count(dagir::ir_attrs::k_label))
44 nn["label"] = std::string(n.attributes.at(dagir::ir_attrs::k_label));
45
46 if (!n.attributes.empty()) {
47 json attrs = json::object();
48 for (const auto& p : n.attributes) {
49 attrs[std::string(p.first)] = std::string(p.second);
50 }
51 nn["attributes"] = std::move(attrs);
52 }
53 out["nodes"].push_back(std::move(nn));
54 }
55
56 // edges
57 out["edges"] = json::array();
58 // For mapping ids to printable names, build a small lookup
59 std::unordered_map<uint64_t, std::string> id_to_name;
60 for (const auto& n : g.nodes) {
61 auto it_name = n.attributes.find(dagir::ir_attrs::k_name);
62 if (it_name != n.attributes.end())
63 id_to_name[n.id] = std::string(it_name->second);
64 else
65 id_to_name[n.id] = std::to_string(n.id);
66 }
67
68 for (const auto& e : g.edges) {
69 json ee;
70 ee["source"] = id_to_name[e.source];
71 ee["target"] = id_to_name[e.target];
72 if (!e.attributes.empty()) {
73 json attrs = json::object();
74 for (const auto& p : e.attributes) attrs[std::string(p.first)] = std::string(p.second);
75 ee["attributes"] = std::move(attrs);
76 }
77 out["edges"].push_back(std::move(ee));
78 }
79
80 if (!g.global_attrs.empty()) {
81 json ga = json::object();
82 for (const auto& p : g.global_attrs) ga[std::string(p.first)] = std::string(p.second);
83 out["graphAttributes"] = std::move(ga);
84 }
85
86 // roots (optional)
87 if (!g.roots.empty()) {
88 json r = json::array();
89 for (const auto& rv : g.roots) r.push_back(std::string(rv));
90 out["roots"] = std::move(r);
91 }
92
93 return out.dump();
94}

96inline dagir::ir_graph from_json(std::string_view sv) {
97 json in = json::parse(std::string(sv));
98 if (!in.contains("schema_version")) throw std::invalid_argument("missing schema_version");
99 int ver = in.at("schema_version").get<int>();
100 if (ver != 1) throw std::invalid_argument("unsupported schema_version");
101
102 if (!in.contains("nodes") || !in.contains("edges"))
103 throw std::invalid_argument("missing required keys: nodes and/or edges");
104
105 dagir::ir_graph g;
106
107 // Map from JSON id string to numeric id
108 std::unordered_map<std::string, uint64_t> id_map;
109 // Track used numeric ids (parsed or assigned) to avoid collisions
110 std::unordered_set<uint64_t> used_ids;
111 uint64_t next_id = 1;
112
113 // Read nodes
114 for (const auto& nn : in.at("nodes")) {
115 if (!nn.contains("id")) throw std::invalid_argument("node missing id");
116 std::string jid = nn.at("id").get<std::string>();
117
118 std::optional<uint64_t> parsed_id;
119 // Try parse as integer and record success explicitly. This avoids using
120 // 0 as a sentinel value so an input id of "0" is preserved.
121 try {
122 size_t idx = 0;
123 unsigned long long v = std::stoull(jid, &idx);
124 if (idx == jid.size()) parsed_id = static_cast<uint64_t>(v);
125 } catch (...) {
126 parsed_id.reset();
127 }
128
129 uint64_t final_id;
130 if (parsed_id.has_value()) {
131 // Reject duplicate numeric ids from input to avoid collisions
132 if (used_ids.find(*parsed_id) != used_ids.end())
133 throw std::invalid_argument("duplicate numeric node id");
134 final_id = *parsed_id;
135 used_ids.insert(final_id);
136 } else {
137 // Find next unused id
138 while (used_ids.find(next_id) != used_ids.end()) ++next_id;
139 final_id = next_id++;
140 used_ids.insert(final_id);
141 }
142 id_map[jid] = final_id;
143
144 dagir::ir_node rn;
145 rn.id = final_id;
146
147 // Label
148 if (nn.contains("label")) {
149 auto sv_label = g.attr_cache.cache_view(nn.at("label").get<std::string>());
150 rn.attributes[g.attr_cache.cache_view(dagir::ir_attrs::k_label)] = sv_label;
151 }
152
153 // Attributes
154 if (nn.contains("attributes")) {
155 for (auto it = nn.at("attributes").begin(); it != nn.at("attributes").end(); ++it) {
156 auto key = g.attr_cache.cache_view(it.key());
157 std::string sval =
158 it.value().is_string() ? it.value().get<std::string>() : it.value().dump();
159 auto val = g.attr_cache.cache_view(sval);
160 rn.attributes[key] = val;
161 }
162 }
163
164 // Preserve original id string when it isn't a plain numeric matching numeric_id
165 if (jid != std::to_string(rn.id)) {
166 rn.attributes[g.attr_cache.cache_view(dagir::ir_attrs::k_id)] = g.attr_cache.cache_view(jid);
167 }
168
169 g.nodes.push_back(std::move(rn));
170 }
171
172 // Read edges
173 for (const auto& ee : in.at("edges")) {
174 if (!ee.contains("source") || !ee.contains("target"))
175 throw std::invalid_argument("edge missing source/target");
176 std::string ssrc = ee.at("source").get<std::string>();
177 std::string stgt = ee.at("target").get<std::string>();
178 if (id_map.find(ssrc) == id_map.end() || id_map.find(stgt) == id_map.end())
179 throw std::invalid_argument("edge references unknown node id");
180
181 dagir::ir_edge re;
182 re.source = id_map[ssrc];
183 re.target = id_map[stgt];
184
185 if (ee.contains("attributes")) {
186 for (auto it = ee.at("attributes").begin(); it != ee.at("attributes").end(); ++it) {
187 auto key = g.attr_cache.cache_view(it.key());
188 std::string sval =
189 it.value().is_string() ? it.value().get<std::string>() : it.value().dump();
190 auto val = g.attr_cache.cache_view(sval);
191 re.attributes[key] = val;
192 }
193 }
194 g.edges.push_back(std::move(re));
195 }
196
197 // global attrs
198 if (in.contains("graphAttributes")) {
199 for (auto it = in.at("graphAttributes").begin(); it != in.at("graphAttributes").end(); ++it) {
200 auto key = g.attr_cache.cache_view(it.key());
201 std::string sval = it.value().is_string() ? it.value().get<std::string>() : it.value().dump();
202 auto val = g.attr_cache.cache_view(sval);
203 g.global_attrs[key] = val;
204 }
205 }
206
207 // roots (optional)
208 if (in.contains("roots") && !in.at("roots").is_null()) {
209 if (!in.at("roots").is_array())
210 throw std::invalid_argument("roots must be an array of strings");
211 for (const auto& rv : in.at("roots")) {
212 if (!rv.is_string()) throw std::invalid_argument("roots elements must be strings");
213 auto rvstr = rv.get<std::string>();
214 g.roots.push_back(g.attr_cache.cache_view(rvstr));
215 }
216 }
217
218 return g;
219}
