40static std::string parenthesize_if_needed(const Expr& child, int parent_prec) {
41 if (child.prec < parent_prec) return std::string("(") + child.s + ")";

54inline void render_expr(std::ostream& os, const ir_graph& g) {
55 using namespace render_expr_detail;
56
57 // Build node lookup for fast access
58 std::unordered_map<std::uint64_t, const ir_node*> node_map;
59 node_map.reserve(g.nodes.size());
60 for (const auto& n : g.nodes) node_map.emplace(n.id, &n);
61
62 // Build adjacency mapping from node id -> (false_target, true_target)
63 struct Targets {
64 std::uint64_t false_t = UINT64_MAX;
65 std::uint64_t true_t = UINT64_MAX;
66 };
67 std::unordered_map<std::uint64_t, Targets> targets;
68 targets.reserve(g.nodes.size());
69
70 for (const auto& e : g.edges) {
71 auto it = e.attributes.find(ir_attrs::k_style);
72 std::string_view style = (it != e.attributes.end()) ? it->second : std::string_view{};
73 if (style == std::string_view("dashed")) {
74 targets[e.source].false_t = e.target;
75 } else if (style == std::string_view("solid")) {
76 targets[e.source].true_t = e.target;
77 }
78 }
79
80 // Memoize computed expressions for node ids
81 std::unordered_map<std::uint64_t, Expr> memo;
82
83 // Recursive lambda to compute expression for node id
84 std::function<Expr(std::uint64_t)> emit = [&](std::uint64_t id) -> Expr {
85 auto mit = memo.find(id);
86 if (mit != memo.end()) return mit->second;
87
88 // Find node
89 auto nit = node_map.find(id);
90 if (nit == node_map.end()) {
91 // Unknown node - treat as constant 0 for safety
92 return memo.emplace(id, Expr{"0", PREC_ATOM}).first->second;
93 }
94 const ir_node& node = *nit->second;
95
96 // Extract label (variable name or terminal value)
97 std::string_view label;
98 auto lit = node.attributes.find(ir_attrs::k_label);
99 if (lit != node.attributes.end())
100 label = lit->second;
101 else
102 label = std::string_view{};
103
104 // Terminal nodes are labelled "0" or "1"
105 if (label == std::string_view("0") || label == std::string_view("1")) {
106 Expr e{std::string(label), PREC_ATOM};
107 memo.emplace(id, e);
108 return e;
109 }
110
111 // Non-terminal: find true/false children
112 Targets t;
113 auto tit = targets.find(id);
114 if (tit != targets.end()) t = tit->second;
115
116 Expr low = (t.false_t == UINT64_MAX) ? Expr{"0", PREC_ATOM} : emit(t.false_t);
117 Expr high = (t.true_t == UINT64_MAX) ? Expr{"0", PREC_ATOM} : emit(t.true_t);
118
119 // If both branches are equal, return branch expression
120 if (low.s == high.s) {
121 memo.emplace(id, high);
122 return high;
123 }
124
125 // If high==1 and low==0 then result is variable
126 std::string var_name = std::string(label);
127 if (high.s == "1" && low.s == "0") {
128 Expr e{var_name, PREC_ATOM};
129 memo.emplace(id, e);
130 return e;
131 }
132 if (high.s == "0" && low.s == "1") {
133 // Equivalent to NOT var
134 Expr e{std::string("NOT ") + var_name, PREC_NOT};
135 memo.emplace(id, e);
136 return e;
137 }
138
139 // General ITE -> (var AND high) OR (NOT var AND low)
140 // Avoid emitting literal terminals '0'/'1' in composed expressions because
141 // the expression parser treats those as variable names. Handle simple
142 // cases to produce cleaner expressions without constant tokens.
143 Expr var_expr{var_name, PREC_ATOM};
144 Expr not_var_expr{std::string("NOT ") + var_name, PREC_NOT};
145
146 // If low is constant 0 then right part is 0 and expression simplifies to
147 // (var AND high). Likewise if high is constant 0 simplify to (NOT var AND low).
148 if (low.s == "0") {
149 std::string left_s = parenthesize_if_needed(var_expr, PREC_AND) + " AND " +
150 parenthesize_if_needed(high, PREC_AND);
151 Expr left{left_s, PREC_AND};
152 memo.emplace(id, left);
153 return left;
154 }
155 if (high.s == "0") {
156 std::string right_s = parenthesize_if_needed(not_var_expr, PREC_AND) + " AND " +
157 parenthesize_if_needed(low, PREC_AND);
158 Expr right{right_s, PREC_AND};
159 memo.emplace(id, right);
160 return right;
161 }
162
163 // If high is constant 1: ITE(var,1,low) == var OR low
164 if (high.s == "1") {
165 std::string a = parenthesize_if_needed(var_expr, PREC_OR);
166 std::string b = parenthesize_if_needed(low, PREC_OR);
167 std::string out_s = a + " OR " + b;
168 Expr out{out_s, PREC_OR};
169 memo.emplace(id, out);
170 return out;
171 }
172
173 // If low is constant 1: ITE(var,high,1) == (NOT var) OR high
174 if (low.s == "1") {
175 std::string a = parenthesize_if_needed(not_var_expr, PREC_OR);
176 std::string b = parenthesize_if_needed(high, PREC_OR);
177 std::string out_s = a + " OR " + b;
178 Expr out{out_s, PREC_OR};
179 memo.emplace(id, out);
180 return out;
181 }
182
183 // Build left = var AND high
184 std::string left_s = parenthesize_if_needed(var_expr, PREC_AND) + " AND " +
185 parenthesize_if_needed(high, PREC_AND);
186 Expr left{left_s, PREC_AND};
187
188 // Build right = NOT var AND low
189 std::string right_s = parenthesize_if_needed(not_var_expr, PREC_AND) + " AND " +
190 parenthesize_if_needed(low, PREC_AND);
191 Expr right{right_s, PREC_AND};
192
193 // Combine: left OR right
194 std::string a_out = parenthesize_if_needed(left, PREC_OR);
195 std::string b_out = parenthesize_if_needed(right, PREC_OR);
196 std::string out_s = a_out + " OR " + b_out;
197 Expr out{out_s, PREC_OR};
198
199 // Cache and return
200 memo.emplace(id, out);
201 return out;
202 };
203
204 // If no roots are present, emit nothing
205 if (g.nodes.empty()) return;
206
207 // Find roots: nodes that are not targets of any edge (graph roots)
208 std::unordered_map<std::uint64_t, int> incoming;
209 for (const auto& n : g.nodes) incoming.emplace(n.id, 0);
210 for (const auto& e : g.edges) {
211 auto it = incoming.find(e.target);
212 if (it != incoming.end()) it->second++;
213 }
214
215 std::vector<std::uint64_t> roots;
216 for (const auto& p : incoming)
217 if (p.second == 0) roots.push_back(p.first);
218
219 // For BDDs produced by the helpers the first root is the diagram root.
220 // If multiple roots are present join them with OR.
221 std::vector<Expr> root_exprs;
222 std::transform(roots.begin(), roots.end(), std::back_inserter(root_exprs), emit);
223
224 if (root_exprs.empty()) return;
225
226 // Combine multiple root expressions with OR
227 std::ostringstream oss;
228 for (size_t i = 0; i < root_exprs.size(); ++i) {
229 if (i) oss << " OR ";
230 // Parenthesize top-level parts if they are lower precedence than OR
231 oss << parenthesize_if_needed(root_exprs[i], PREC_OR);
232 }
233
234 os << oss.str();
235}
