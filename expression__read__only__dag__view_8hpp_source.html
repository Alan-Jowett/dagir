78 auto children(const handle& h) const {

86 if (p_or->left) out.push_back(expression_edge{handle{p_or->left.get()}});
87 if (p_or->right) out.push_back(expression_edge{handle{p_or->right.get()}});
88 } else if (auto p_xor = std::get_if<my_xor>(h.ptr)) {

90 if (p_xor->right) out.push_back(expression_edge{handle{p_xor->right.get()}});
91 } else if (auto p_not = std::get_if<my_not>(h.ptr)) {
92 if (p_not->expr) out.push_back(expression_edge{handle{p_not->expr.get()}});
