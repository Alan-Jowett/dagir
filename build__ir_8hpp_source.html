87template <dagir::concepts::read_only_dag_view View, class NodePolicy, class EdgePolicy>
88 requires dagir::concepts::node_attributor<NodePolicy, View>
89ir_graph build_ir(const View& view, NodePolicy&& node_policy, EdgePolicy&& edge_attr,
90 bool detect_cycles_flag = false) {
91 using H = typename View::handle;
92 using key_t = std::uint64_t;
93
94 ir_graph graph;
95
96 // Detect cycles if requested and get traversal order
97 dagir::cycle_info cycle_detection;
98 std::vector<H> topo;
99
100 if (detect_cycles_flag) {
101 // Use combined function to detect cycles and get traversal in one pass
102 cycle_detection = dagir::detect_cycles_with_traversal(view, topo);
103
104 // Add graph-level cycle flag
105 if (cycle_detection.has_cycles) {
106 std::string_view key = graph.attr_cache.cache_view(std::string(ir_attrs::k_has_cycles));
107 std::string_view val = graph.attr_cache.cache_view("true");
108 graph.global_attrs[key] = val;
109 }
110 } else {
111 // Use Kahn's algorithm for backward compatibility (throws on cycles)
112 topo = kahn_topological_order(view);
113 }
114
115 graph.nodes.reserve(topo.size());
116
117 // First, create nodes (memoized) using label policy
118 for (std::size_t idx = 0; idx < topo.size(); ++idx) {
119 const H& h = topo[idx];
120 key_t k = h.stable_key();
121
122 // Optionally guard traversal for this node
123 if constexpr (requires(const View& v, H hh) { v.start_guard(hh); }) {
124 auto guard = view.start_guard(h);
125 (void)guard;
126 }
127
128 ir_node n;
129 n.id = k;
130
131 // Default canonical name assigned in topological order; policies must
132 // be node-attributors producing `dagir::ir_attr_map` that will populate
133 // `n.attributes`. We prefer attribute-provided values; otherwise the
134 // default name is used and a label from the stable key is written.
135 auto attributes = std::invoke(node_policy, view, h);
136
137 // Copy the returned attributes into the node. We must ensure that
138 // `std::string_view` keys/values stored in `ir_graph` remain valid for
139 // the lifetime of the graph. Use `graph.attr_cache` to obtain cached
140 // string_views and insert those into the node's attribute map.
141 for (const auto& [attr_key, attr_value] : attributes) {
142 std::string key_copy(attr_key);
143 std::string val_copy(attr_value);
144 std::string_view ksv = graph.attr_cache.cache_view(key_copy);
145 std::string_view vsv = graph.attr_cache.cache_view(val_copy);
146 n.attributes[ksv] = vsv;
147 }
148
149 if (!n.attributes.count(ir_attrs::k_name)) {
150 std::string name_val = std::format("node{:03}", idx);
151 n.attributes[graph.attr_cache.cache_view(ir_attrs::k_name)] =
152 graph.attr_cache.cache_view(name_val);
153 }
154 if (!n.attributes.count(ir_attrs::k_label)) {
155 std::string label_val = std::to_string(k);
156 n.attributes[graph.attr_cache.cache_view(ir_attrs::k_label)] =
157 graph.attr_cache.cache_view(label_val);
158 }
159
160 // Add cycle group if node is part of an SCC
161 if (detect_cycles_flag && cycle_detection.cycle_groups.count(k)) {
162 std::size_t scc_id = cycle_detection.cycle_groups.at(k);
163 std::string scc_str = std::to_string(scc_id);
164 std::string_view key = graph.attr_cache.cache_view(std::string(ir_attrs::k_cycle_group));
165 std::string_view val = graph.attr_cache.cache_view(scc_str);
166 n.attributes[key] = val;
167 }
168
169 graph.nodes.push_back(std::move(n));
170 }
171
172 // Now collect edges; reserve approximate size by summing child counts
173 // Reserve an approximate size for edges by summing child counts using
174 // standard algorithms. Using std::accumulate makes the intent clearer
175 // and satisfies cppcheck's `useStlAlgorithm` suggestion.
176 std::size_t est_edges = std::accumulate(
177 topo.begin(), topo.end(), std::size_t{0}, [&view](std::size_t acc, const H& h) {
178 return acc + static_cast<std::size_t>(std::ranges::distance(view.children(h)));
179 });
180 graph.edges.reserve(est_edges);
181
182 for (const H& parent : topo) {
183 key_t pk = parent.stable_key();
184 for (auto const& edge_like : view.children(parent)) {
185 H child = build_ir_extract_child<H>(edge_like);
186 key_t ck = child.stable_key();
187
188 ir_edge ie;
189 ie.source = pk;
190 ie.target = ck;
191
192 // Determine attributes via flexible invocation forms. Use a
193 // constexpr-dispatched lambda returning whatever type the policy
194 // produces so we can iterate the resulting name/value range
195 // generically (it may be an owning container of `std::string`s or
196 // a view-based map). The lambda's return type is deduced from the
197 // selected branch at compile-time.
198 auto edge_attrs = [&]() {
199 if constexpr (std::invocable<EdgePolicy, const View&, const H&,
200 const decltype(edge_like)&>) {
201 return std::invoke(edge_attr, view, parent, edge_like);
202 } else if constexpr (std::invocable<EdgePolicy, const View&, const H&, const H&>) {
203 return std::invoke(edge_attr, view, parent, child);
204 } else if constexpr (std::invocable<EdgePolicy, const H&, const decltype(edge_like)&>) {
205 return std::invoke(edge_attr, parent, edge_like);
206 } else if constexpr (std::invocable<EdgePolicy, const H&, const H&>) {
207 return std::invoke(edge_attr, parent, child);
208 } else {
209 return dagir::ir_attr_map{};
210 }
211 }();
212
213 // Cache keys/values from the returned map into the graph's attr_cache
214 for (const auto& [ek, ev] : edge_attrs) {
215 std::string key_copy(ek);
216 std::string val_copy(ev);
217 std::string_view ksv = graph.attr_cache.cache_view(key_copy);
218 std::string_view vsv = graph.attr_cache.cache_view(val_copy);
219 ie.attributes[ksv] = vsv;
220 }
221
222 // Mark back-edges if cycle detection is enabled
223 if (detect_cycles_flag && cycle_detection.has_cycles) {
224 auto edge_key = std::make_pair(pk, ck);
225 if (cycle_detection.back_edges.count(edge_key) && cycle_detection.back_edges.at(edge_key)) {
226 std::string_view key =
227 graph.attr_cache.cache_view(std::string(ir_attrs::k_is_cycle_back_edge));
228 std::string_view val = graph.attr_cache.cache_view("true");
229 ie.attributes[key] = val;
230 }
231 }
232
233 graph.edges.push_back(std::move(ie));
234 }
235 }
236
237 return graph;
238}

246template <dagir::concepts::read_only_dag_view View>
247ir_graph build_ir(const View& view) {
248 // The default attributor must ensure any `std::string_view` it returns
249 // remains valid until the outer `build_ir` call consumes and caches
250 // the values. To guarantee this, keep a local container of owning
251 // `std::string`s and return `std::string_view`s that refer into that
252 // container. The container lives for the duration of this function
253 // invocation and thus prevents dangling views produced from
254 // prvalue temporaries.
255 std::vector<std::string> __default_label_storage;
256 auto node_attr = [&__default_label_storage](auto const& /*view*/,
257 auto const& h) -> dagir::ir_attr_map {
258 dagir::ir_attr_map m;
259 __default_label_storage.emplace_back(std::format("{}", h.stable_key()));
260 m.emplace(ir_attrs::k_label, std::string_view(__default_label_storage.back()));
261 return m;
262 };
263 auto edge_attr = [](auto&&...) -> dagir::ir_attr_map { return {}; };
264 return build_ir(view, node_attr, edge_attr);
265}
