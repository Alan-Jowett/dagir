30struct pair_hash {
31 template <class T1, class T2>
32 std::size_t operator()(const std::pair<T1, T2>& p) const {
33 auto h1 = std::hash<T1>{}(p.first);
34 auto h2 = std::hash<T2>{}(p.second);
35 return h1 ^ (h2 << 1);
36 }
37};

74template <dagir::concepts::read_only_dag_view View>
75std::vector<typename View::handle> kahn_topological_order(const View& view) {
76 using H = typename View::handle;
77 using key_t = std::uint64_t;
78
79 std::unordered_map<key_t, std::size_t> indeg;
80 std::unordered_map<key_t, H> handle_of;
81 std::unordered_set<key_t> seen;
82 std::vector<key_t> nodes; // all discovered node keys (for iteration)
83
84 // helper to extract a child handle from a range element (edge or handle).
85 auto extract_child = []<class E>(const E& e) -> H {
86 if constexpr (std::convertible_to<E, H>) {
87 return static_cast<H>(e);
88 } else {
89 return e.target();
90 }
91 };
92
93 // BFS/stack from roots to discover reachable nodes and compute indegrees
94 std::vector<H> work;
95 for (auto const& r : view.roots()) {
96 H h = r;
97 key_t k = h.stable_key();
98 if (seen.insert(k).second) {
99 handle_of.emplace(k, h);
100 indeg.try_emplace(k, 0);
101 nodes.push_back(k);
102 work.push_back(h);
103 }
104 }
105
106 for (std::size_t i = 0; i < work.size(); ++i) {
107 H cur = work[i];
108 for (auto const& edge_like : view.children(cur)) {
109 H child = extract_child(edge_like);
110 key_t ck = child.stable_key();
111 // ensure child is in maps
112 handle_of.try_emplace(ck, child);
113 auto [it, inserted] = indeg.try_emplace(ck, 0);
114 ++(it->second);
115
116 if (seen.insert(ck).second) {
117 nodes.push_back(ck);
118 work.push_back(child);
119 }
120 }
121 }
122
123 // Kahn: push all zero-indegree nodes
124 std::queue<key_t> q;
125 for (auto const& k : nodes) {
126 if (indeg[k] == 0) q.push(k);
127 }
128
129 std::vector<H> order;
130 order.reserve(nodes.size());
131
132 while (!q.empty()) {
133 key_t k = q.front();
134 q.pop();
135 order.push_back(handle_of.at(k));
136
137 H h = handle_of.at(k);
138 for (auto const& edge_like : view.children(h)) {
139 H child = extract_child(edge_like);
140 key_t ck = child.stable_key();
141 auto it = indeg.find(ck);
142 if (it == indeg.end()) continue; // child outside discovered set
143 if (--(it->second) == 0) q.push(ck);
144 }
145 }
146
147 if (order.size() != nodes.size())
148 throw std::runtime_error("kahn_topological_order: cycle detected in reachable graph");
149
150 return order;
151}

170template <dagir::concepts::read_only_dag_view View>
171cycle_info detect_cycles(const View& view) {
172 using H = typename View::handle;
173 using key_t = std::uint64_t;
174
175 cycle_info result;
176
177 // DFS state tracking
178 enum class State { Unvisited, Visiting, Visited };
179 std::unordered_map<key_t, State> state;
180 std::unordered_map<key_t, H> handle_of;
181
182 // Tarjan's algorithm state
183 std::unordered_map<key_t, std::size_t> dfs_num;
184 std::unordered_map<key_t, std::size_t> low_link;
185 std::vector<key_t> stack;
186 std::unordered_set<key_t> on_stack;
187 std::size_t counter = 0;
188
189 // Helper to extract child handle
190 auto extract_child = [](const auto& edge_like) -> H {
191 if constexpr (std::convertible_to<decltype(edge_like), H>) {
192 return static_cast<H>(edge_like);
193 } else {
194 return edge_like.target();
195 }
196 };
197
198 // Tarjan's DFS to find SCCs and back-edges
199 std::function<void(H)> tarjan_dfs = [&](H node) {
200 key_t k = node.stable_key();
201
202 state[k] = State::Visiting;
203 dfs_num[k] = low_link[k] = counter++;
204 stack.push_back(k);
205 on_stack.insert(k);
206 handle_of[k] = node;
207
208 for (auto const& edge_like : view.children(node)) {
209 H child = extract_child(edge_like);
210 key_t ck = child.stable_key();
211
212 handle_of.try_emplace(ck, child);
213
214 auto it = state.find(ck);
215 if (it == state.end() || it->second == State::Unvisited) {
216 // Tree edge - recurse
217 tarjan_dfs(child);
218 low_link[k] = std::min(low_link[k], low_link[ck]);
219 } else if (it->second == State::Visiting) {
220 // Back-edge - cycle detected
221 result.has_cycles = true;
222 result.back_edges[{k, ck}] = true;
223 low_link[k] = std::min(low_link[k], dfs_num[ck]);
224 }
225 // else: cross or forward edge in visited node
226 }
227
228 state[k] = State::Visited;
229
230 // Found SCC root?
231 if (low_link[k] == dfs_num[k]) {
232 std::size_t scc_id = result.num_sccs++;
233 key_t w;
234 do {
235 w = stack.back();
236 stack.pop_back();
237 on_stack.erase(w);
238 result.cycle_groups[w] = scc_id;
239 } while (w != k);
240 }
241 };
242
243 // Run DFS from all roots
244 for (auto const& r : view.roots()) {
245 H root = r;
246 key_t k = root.stable_key();
247 if (state.find(k) == state.end()) {
248 tarjan_dfs(root);
249 }
250 }
251
252 return result;
253}

272cycle_info detect_cycles_with_traversal(const View& view,
273 std::vector<typename View::handle>& out_order) {
274 using H = typename View::handle;
275 using key_t = std::uint64_t;
276
277 cycle_info result;
278 out_order.clear();

292 // Helper to extract child handle

309 handle_of[k] = node;
310
311 // Optionally guard traversal for this node
312 if constexpr (requires(const View& v, H hh) { v.start_guard(hh); }) {
313 auto guard = view.start_guard(node);
314 (void)guard;
315 }
316

318 H child = extract_child(edge_like);
319 key_t ck = child.stable_key();
320

322

324 if (it == state.end() || it->second == State::Unvisited) {
325 // Tree edge - recurse
326 tarjan_dfs(child);
327 low_link[k] = std::min(low_link[k], low_link[ck]);
328 } else if (it->second == State::Visiting) {
329 // Back-edge - cycle detected
330 result.has_cycles = true;
331 result.back_edges[{k, ck}] = true;
332 low_link[k] = std::min(low_link[k], dfs_num[ck]);
333 }
334 // else: cross or forward edge in visited node
335 }
336
337 state[k] = State::Visited;
338
339 // Add to traversal order in postorder
340 out_order.push_back(node);
341
342 // Found SCC root?
343 if (low_link[k] == dfs_num[k]) {
344 std::size_t scc_id = result.num_sccs++;
345 key_t w;
346 do {
347 w = stack.back();
348 stack.pop_back();
349 on_stack.erase(w);
350 result.cycle_groups[w] = scc_id;
351 } while (w != k);
352 }
353 };
354
355 // Run DFS from all roots
356 for (auto const& r : view.roots()) {
357 H root = r;
358 key_t k = root.stable_key();
359 if (state.find(k) == state.end()) {
360 tarjan_dfs(root);
361 }
362 }
363
364 // Reverse to get topological-like order
365 std::reverse(out_order.begin(), out_order.end());
366
367 return result;
368}

387template <dagir::concepts::read_only_dag_view View>
388std::vector<typename View::handle> dfs_traversal_order(const View& view) {
389 using H = typename View::handle;
390 using key_t = std::uint64_t;
391
392 std::unordered_set<key_t> visited;
393 std::unordered_map<key_t, H> handle_of;
394 std::vector<H> order;
395
396 // Helper to extract child handle
397 auto extract_child = [](const auto& edge_like) -> H {
398 if constexpr (std::convertible_to<decltype(edge_like), H>) {
399 return static_cast<H>(edge_like);
400 } else {
401 return edge_like.target();
402 }
403 };
404
405 // DFS with postorder collection
406 std::function<void(H)> dfs = [&](H node) {
407 key_t k = node.stable_key();
408
409 if (!visited.insert(k).second) {
410 return; // Already visited
411 }
412
413 handle_of[k] = node;
414
415 // Optionally guard traversal for this node
416 if constexpr (requires(const View& v, H hh) { v.start_guard(hh); }) {
417 auto guard = view.start_guard(node);
418 (void)guard;
419 }
420
421 // Visit children first (postorder)
422 for (auto const& edge_like : view.children(node)) {
423 H child = extract_child(edge_like);
424 key_t ck = child.stable_key();
425
426 if (!visited.count(ck)) {
427 dfs(child);
428 }
429 }
430
431 // Add node after visiting children
432 order.push_back(node);
433 };
434
435 // Start DFS from all roots
436 for (auto const& r : view.roots()) {
437 H root = r;
438 key_t k = root.stable_key();
439 if (!visited.count(k)) {
440 dfs(root);
441 }
442 }
443
444 // Reverse to get topological-like order for DAGs
445 std::reverse(order.begin(), order.end());
446
447 return order;
448}

468template <dagir::concepts::read_only_dag_view View, class R, class Combiner>
469auto postorder_fold(const View& view, Combiner combiner) -> std::unordered_map<std::uint64_t, R> {
470 using H = typename View::handle;
471 using key_t = std::uint64_t;
472
473 auto topo = kahn_topological_order(view);
474 std::unordered_map<key_t, R> results;
475 results.reserve(topo.size());
476
477 // helper to extract child handle (same as above)
478 auto extract_child = []<class E>(const E& e) -> H {
479 if constexpr (std::convertible_to<E, H>) {
480 return static_cast<H>(e);
481 } else {
482 return e.target();
483 }
484 };
485
486 // Process in reverse topological order: children before parents
487 for (auto it = topo.rbegin(); it != topo.rend(); ++it) {
488 H node = *it;
489 key_t k = node.stable_key();
490
491 // collect child results in order observed
492 std::vector<R> child_vals;
493 for (auto const& edge_like : view.children(node)) {
494 H child = extract_child(edge_like);
495 key_t ck = child.stable_key();
496 auto found = results.find(ck);
497 if (found != results.end())
498 child_vals.push_back(found->second);
499 else
500 child_vals.push_back(R{}); // absent->default constructed result
501 }
502
503 R res = std::invoke(combiner, view, node, std::span(child_vals));
504 results.emplace(k, std::move(res));
505 }
506
507 return results;
508}
