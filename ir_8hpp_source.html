78inline bool operator<(ir_node const& a, ir_node const& b) {
79 const auto a_it = a.attributes.find(ir_attrs::k_name);
80 const auto b_it = b.attributes.find(ir_attrs::k_name);
81 const bool a_has = (a_it != a.attributes.end());
82 const bool b_has = (b_it != b.attributes.end());
83 if (a_has && b_has) {
84 const std::string_view a_name = a_it->second;
85 const std::string_view b_name = b_it->second;
86 if (a_name != b_name) return a_name < b_name;
87 return a.id < b.id;
88 }
89 if (a_has && !b_has) return true; // named nodes come before unnamed
90 if (!a_has && b_has) return false; // unnamed after named
91 return a.id < b.id;
92}

130inline bool operator<(ir_edge const& a, ir_edge const& b) {
131 // Compare by source id, then target id, then by style attribute (if present).
132 const auto a_style_it = a.attributes.find(ir_attrs::k_style);
133 const auto b_style_it = b.attributes.find(ir_attrs::k_style);
134 const std::string_view a_style =
135 (a_style_it != a.attributes.end()) ? a_style_it->second : std::string_view{};
136 const std::string_view b_style =
137 (b_style_it != b.attributes.end()) ? b_style_it->second : std::string_view{};
138 return std::tie(a.source, a.target, a_style) < std::tie(b.source, b.target, b_style);
139}
