97struct expression_edge_attributor {
98 using handle = typename expression_read_only_dag_view::handle;
99
100 std::vector<std::pair<std::string_view, std::string_view>> operator()(
101 const expression_read_only_dag_view& /*view*/, const handle& parent,
102 const handle& child) const {
103 std::vector<std::pair<std::string_view, std::string_view>> out;
104 if (!parent.ptr) return out;
105 // For binary operators, label edges left/right. Keep some styles
106 // (e.g. bold for AND, dashed for NOT) but do not set edge colors.
107 const std::string_view key_label = dagir::ir_attrs::k_label;
108 if (auto p_and = std::get_if<my_and>(parent.ptr)) {
109 if (p_and->left && p_and->left.get() == child.ptr)
110 out.emplace_back(key_label, std::string_view("L"));
111 else if (p_and->right && p_and->right.get() == child.ptr)
112 out.emplace_back(key_label, std::string_view("R"));
113 } else if (auto p_or = std::get_if<my_or>(parent.ptr)) {
114 if (p_or->left && p_or->left.get() == child.ptr)
115 out.emplace_back(key_label, std::string_view("L"));
116 else if (p_or->right && p_or->right.get() == child.ptr)
117 out.emplace_back(key_label, std::string_view("R"));
118 } else if (auto p_xor = std::get_if<my_xor>(parent.ptr)) {
119 if (p_xor->left && p_xor->left.get() == child.ptr)
120 out.emplace_back(key_label, std::string_view("L"));
121 else if (p_xor->right && p_xor->right.get() == child.ptr)
122 out.emplace_back(key_label, std::string_view("R"));
123 }
124 return out;
125 }
